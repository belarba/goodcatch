# Pista (novo jogo) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A standalone daily deduction page at `novo-jogo/index.html`: find the fish of the day's three habits with at most six casts, each cast answering only how many habits the square meets.

**Architecture:** One static page, copied pieces from `index.html` (PRNG, sprite renderer, tokens, `track`, `share`), with the pure logic between `// @gen-start` and `// @gen-end`. `tools/pista-check.mjs` evaluates that block in Node and asserts the generator's invariants over 60 days, including par 4 and the worst consistent player within six casts.

**Tech Stack:** static HTML/CSS/JS, Node for the check (no dependencies).

**Spec:** `docs/superpowers/specs/2026-09-30-pista-design.md`

## Global Constraints

- Branch `novo-jogo`; nothing is pushed to `main` without the owner's go. `index.html` is not touched.
- 7×7, island 2×2, shallow = Chebyshev distance ≤ 2 to the island, exactly 12 of 16 combinations present.
- Six casts; ★★★ ≤ 4, ★★ 5, ★ 6; lost = `X/6`, no stars.
- Record `goodcatch:<date>:pista` (array of cast square keys), localStorage in try/catch.
- Every string in `STR.en` and `STR.pt`. Comments only on critical lines. Numbers from commands.
- GoatCounter `belarba`, skipped on localhost: page view, then `pista/caught-N` or `pista/lost` once.

**Interfaces (produced, used by Task 2):**

- `genSea(date) → {grid, sun, fish, present, tries}`; `grid[r][c]` is `{island:true}` or `{b:0..3, d:0|1, l:0|1}` (d 1 = deep, l 1 = shade); `fish` is `{b,d,l}`; `present` is the sorted combo list
- `combo(x) = x.b*4 + x.d*2 + x.l`; `bites(x, fish) → 0..3`
- `stars(n) → 3|2|1` for a catch in n casts

```js
// Invariants of the novo-jogo sea over 60 days: deterministic, exactly 12 combinations, the fish among them,
// par 4 and a consistent player always catching within six casts.
import { readFileSync } from 'node:fs';
const html = readFileSync(new URL('../novo-jogo/index.html', import.meta.url), 'utf8');
const a = html.indexOf('// @gen-start'), b = html.indexOf('// @gen-end');
if (a < 0 || b < a) throw new Error('@gen markers not found');
const G = new Function(`${html.slice(a, b)}; return {genSea, combo, bites, stars, N};`)();

const F = [...Array(16).keys()].map(q => [...Array(16).keys()].map(s => G.bites(
  { b: q >> 2, d: (q >> 1) & 1, l: q & 1 }, { b: s >> 2, d: (s >> 1) & 1, l: s & 1 })));
const split = (q, S) => { const m = new Map(); for (const s of S) { const f = F[q][s]; if (!m.has(f)) m.set(f, []); m.get(f).push(s); } return m; };
function par(S, P, memo = new Map()) {
  if (S.length === 1) return 1; const k = S.join(); if (memo.has(k)) return memo.get(k); let best = 99;
  for (const q of P) { const m = split(q, S); if (m.size === 1 && !m.has(3)) continue;
    let w = 0; for (const [f, p] of m) if (f !== 3) w = Math.max(w, par(p, P, memo)); best = Math.min(best, 1 + w); }
  memo.set(k, best); return best;
}
// Adversary picks both the consistent cast and the answer.
function worst(S, memo = new Map()) {
  if (S.length === 1) return 1; const k = S.join(); if (memo.has(k)) return memo.get(k); let w = 1;
  for (const q of S) for (const [f, p] of split(q, S)) if (f !== 3) w = Math.max(w, 1 + worst(p, memo));
  memo.set(k, w); return w;
}
if (G.bites({ b: 1, d: 1, l: 0 }, { b: 1, d: 0, l: 0 }) !== 2) throw new Error('bites: coral deep sun vs coral shallow sun must be 2');
if (JSON.stringify([4, 5, 6].map(G.stars)) !== '[3,2,1]') throw new Error('stars: 4/5/6 casts must give 3/2/1');
const tries = [];
for (let i = 0; i < 60; i++) {
  const d = new Date(2026, 9, 1 + i), date = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`;
  const s = G.genSea(date);
  if (JSON.stringify(s) !== JSON.stringify(G.genSea(date))) throw new Error(`${date}: not deterministic`);
  const cells = s.grid.flat().filter(x => !x.island), present = [...new Set(cells.map(G.combo))].sort((x, y) => x - y);
  if (s.grid.flat().length !== G.N * G.N || cells.length !== G.N * G.N - 4) throw new Error(`${date}: expected 7x7 with a 2x2 island`);
  if (present.length !== 12 || present.join() !== s.present.join()) throw new Error(`${date}: ${present.length} combinations present`);
  if (!present.includes(G.combo(s.fish))) throw new Error(`${date}: the fish is not on the map`);
  const p = par(present, present), w = worst(present);
  if (p !== 4) throw new Error(`${date}: par ${p}, the stars assume 4`);
  if (w > 6) throw new Error(`${date}: a consistent player can need ${w} casts`);
  tries.push(s.tries);
}
tries.sort((x, y) => x - y);
console.log(`pista ok: 60 days, generator tries median ${tries[30]} max ${tries.at(-1)}`);
```

```js
// @gen-start
const N=7, BOTTOMS=['areia','coral','pedra','alga'], DIRS=[[-1,0],[0,1],[1,0],[0,-1]];
function seedFrom(s){ /* verbatim from index.html */ }
function mulberry(a){ /* verbatim from index.html */ }
const key=(r,c)=>r*100+c;
const combo=x=>x.b*4+x.d*2+x.l;
const bites=(x,f)=>(x.b===f.b)+(x.d===f.d)+(x.l===f.l);
const stars=n=>n<=4?3:n===5?2:1;
function genSea(date){
  for(let tries=1;;tries++){
    const rng=mulberry(seedFrom(`${date}:pista${tries>1?':'+tries:''}`)), ri=n=>Math.floor(rng()*n);
    const r0=1+ri(4), c0=1+ri(4), sun=ri(4), [sr,sc]=DIRS[sun];
    const pts=Array.from({length:6},(_,i)=>({r:rng()*N,c:rng()*N,b:i<4?i:ri(4)}));
    const grid=[];
    for(let r=0;r<N;r++){grid.push([]);for(let c=0;c<N;c++){
      if(r>=r0&&r<=r0+1&&c>=c0&&c<=c0+1){grid[r].push({island:true});continue}
      const dr=Math.max(r0-r,0,r-r0-1), dc=Math.max(c0-c,0,c-c0-1);
      const along=sr?(sr<0?r-(r0+1):r0-r):(sc>0?c0-c:c-(c0+1)), across=sr?dc:dr;
      let b=0,bd=1e9;for(const p of pts){const q=(p.r-r-.5)**2+(p.c-c-.5)**2;if(q<bd){bd=q;b=p.b}}
      grid[r].push({b,d:Math.max(dr,dc)<=2?0:1,l:along>0&&across<=Math.floor(along/2)?1:0});
    }}
    const present=[...new Set(grid.flat().filter(x=>!x.island).map(combo))].sort((x,y)=>x-y);
    if(present.length===12){const f=present[ri(12)];return {grid,sun,fish:{b:f>>2,d:(f>>1)&1,l:f&1},present,tries}}
  }
}
// @gen-end
```
